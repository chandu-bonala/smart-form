interface AuthInputProps {
  label: string;
  placeholder: string;
  value: string;
  onChangeText: (text: string) => void;
  error?: string;
  keyboardType?: 'default' | 'email-address' | 'numeric' | 'phone-pad';
  autoCapitalize?: 'none' | 'sentences' | 'words' | 'characters';
  secureTextEntry?: boolean;
  editable?: boolean;
}

const AuthInput: React.FC<AuthInputProps> = ({
  label,
  placeholder,
  value,
  onChangeText,
  error,
  keyboardType = 'default',
  autoCapitalize = 'sentences',
  secureTextEntry = false,
  editable = true,
}) => (
  <View style={styles.inputGroup}>
    <Text style={styles.inputLabel}>{label}</Text>
    <View style={[styles.inputShell, error ? styles.inputShellError : null]}>
      <TextInput
        value={value}
        onChangeText={onChangeText}
        placeholder={placeholder}
        placeholderTextColor={palette.muted}
        keyboardType={keyboardType}
        autoCapitalize={autoCapitalize}
        secureTextEntry={secureTextEntry}
        editable={editable}
        style={styles.inputField}
      />
    </View>
    {error ? <Text style={styles.inputErrorText}>{error}</Text> : null}
  </View>
);

interface AuthButtonProps {
  label: string;
  onPress: () => void;
  variant?: 'primary' | 'ghost';
  isLoading?: boolean;
  disabled?: boolean;
}

const AuthButton: React.FC<AuthButtonProps> = ({
  label,
  onPress,
  variant = 'primary',
  isLoading = false,
  disabled = false,
}) => {
  const isDisabled = disabled || isLoading;
  const isGhost = variant === 'ghost';

  return (
    <TouchableOpacity
      style={[
        styles.button,
        isGhost ? styles.buttonGhost : styles.buttonPrimary,
        isDisabled ? styles.buttonDisabled : null,
      ]}
      onPress={onPress}
      disabled={isDisabled}
      activeOpacity={0.88}
    >
      {isLoading ? (
        <ActivityIndicator color={isGhost ? palette.primary : palette.white} size="small" />
      ) : (
        <Text style={[styles.buttonText, isGhost ? styles.buttonTextGhost : null]}>{label}</Text>
      )}
    </TouchableOpacity>
  );
};

export const LoginScreen: React.FC<LoginScreenProps> = ({ navigation }) => {
  const { login, isLoading, error, clearError } = useAuth();
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [emailError, setEmailError] = useState('');
  const [passwordError, setPasswordError] = useState('');

  const validateForm = (): boolean => {
    let isValid = true;

    const emailErr = getEmailErrorMessage(email);
    if (emailErr) {
      setEmailError(emailErr);
      isValid = false;
    } else {
      setEmailError('');
    }

    const passwordErr = getPasswordErrorMessage(password);
    if (passwordErr) {
      setPasswordError(passwordErr);
      isValid = false;
    } else {
      setPasswordError('');
    }

    return isValid;
  };

  const handleLogin = async () => {
    clearError();
    if (!validateForm()) return;

    try {
      await login(email, password);
    } catch (err) {
      console.error('Login error:', err);
    }
  };

  return (
    <KeyboardAvoidingView
      behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
      style={styles.container}
    >
      <ScrollView
        contentContainerStyle={styles.scrollContent}
        keyboardShouldPersistTaps="handled"
        showsVerticalScrollIndicator={false}
      >
        <View style={styles.brandHeader}>
          <View style={styles.logoMark}>
            <Text style={styles.logoText}>SF</Text>
          </View>
          <View style={styles.brandCopy}>
            <Text style={styles.brandName}>SmartForm</Text>
            <Text style={styles.brandLine}>Designed response operations</Text>
          </View>
        </View>

        <View style={styles.headingBlock}>
          <Text style={styles.title}>Welcome back to command center</Text>
          <Text style={styles.subtitle}>
            Sign in to manage forms, responses, analytics, and team activity.
          </Text>
        </View>

        <View style={styles.card}>
          {error ? (
            <View style={styles.errorAlert}>
              <Text style={styles.errorAlertText}>{error}</Text>
            </View>
          ) : null}

          <GoogleAuthButton disabled={isLoading} />

          <AuthInput
            label="Email address"
            placeholder="you@example.com"
            keyboardType="email-address"
            autoCapitalize="none"
            value={email}
            onChangeText={(text) => {
              setEmail(text);
              if (emailError) setEmailError('');
            }}
            error={emailError}
            editable={!isLoading}
          />

          <AuthInput
            label="Password"
            placeholder="Enter your password"
            secureTextEntry
            value={password}
            onChangeText={(text) => {
              setPassword(text);
              if (passwordError) setPasswordError('');
            }}
            error={passwordError}
            editable={!isLoading}
          />

          <AuthButton
            label="Sign in"
            onPress={handleLogin}
            isLoading={isLoading}
            disabled={isLoading}
          />

          <View style={styles.switchRow}>
            <Text style={styles.switchText}>Need an account?</Text>
            <TouchableOpacity
              onPress={() => navigation.replace('Register')}
              disabled={isLoading}
              activeOpacity={0.75}
            >
              <Text style={styles.switchLink}>Create one</Text>
            </TouchableOpacity>
          </View>
        </View>
      </ScrollView>
    </KeyboardAvoidingView>
  );
};

  const [error, setError] = useState<string | null>(null);

  const loadDashboard = useCallback(async () => {
    if (!user) return;

    try {
      setError(null);
      const [dashboard, formResponse] = await Promise.all([
        analyticsService.getDashboard(),
        formsService.list(),
      ]);
      setAnalytics(dashboard);
      setForms(formResponse.items);
    } catch (err)      setError(getErrorMessage(err));
    } finally {
      setLoading(false);
      setRefreshing(false);
    }
  }, [user]);

  useEffect(() => {
    loadDashboard();
  }, [loadDashboard]);

  useEffect(() => {
    const refreshTimer = setInterval(loadDashboard, AUTO_REFRESH_INTERVAL_MS);
    return () => clearInterval(refreshTimer);
  }, [loadDashboard]);

  const recentForms = forms.slice(0, 5);
  const publishedForms = forms.filter((form) => form.status.toLowerCase() === 'published').length;
  const draftForms = forms.filter((form) => form.status.toLowerCase() === 'draft').length;
  const hiddenForms = forms.filter((form) => form.is_hidden).length;
  const totalResponses = analytics?.total_responses ?? 0;
  const completionRate = Math.min(100, Math.round(analytics?.completion_rate ?? 0));
  const avgTime = Math.round(analytics?.average_completion_time ?? 0);
  const trendRows = useMemo(() => {
    const daily = analytics?.daily_responses ?? [];
    const monthly = analytics?.monthly_responses ?? [];
    return (daily.length ? daily : monthly).slice(-10);

  const maxTrend = Math.max(1, ...trendRows.map((item) => item.count));
  const firstName = user?.name?.split(' ')[0] ?? 'Creator';

    ? user.name.split(' ').map((part) => part.charAt(0)).join('').slice(0, 2).toUpperCase()
    : 'SF';

  const onRefresh = () => {
    setRefreshing(true);
    loadDashboard();
  };

  if (!user) {
    return (
      <SafeAreaView style={styles.container}>
        <View style={styles.centered}>
          <ActivityIndicator color={ui.teal} />
        </View>
      </SafeAreaView>
    );
  }

  return (
    <SafeAreaView style={styles.container}>
      <ScrollView
        contentContainerStyle={styles.content}
        refreshControl={<RefreshControl refreshing={refreshing} onRefresh={onRefresh} colors={[ui.teal]} />}
        showsVerticalScrollIndicator={false}
      >
        <View style={styles.shell}>
          <View style={styles.topBar}>
            <View style={styles.brandRow}>
              <View style={styles.logo}>
                <Text style={styles.logoText}>SF</Text>

          </View>

          {error ? <Message title="Live data paused" text={error} /> : null}

              </TouchableOpacity>

            <Metric label="Completion" value={`${completionRate}%`} helper={completionRate >= 70 ? 'Healthy respondent flow' : 'Needs form tuning'} tone={completionRate >= 70 ? 'teal' : 'amber'} loading={loading} />

                <ProgressLine label="Completion quality" value={completionRate} tone={completionRate >= 70 ? 'teal' : 'amber'} />
